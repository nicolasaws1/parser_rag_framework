# Extrator híbrido SB100 — Squad 2
# YOLO roteia · Chandra OCR 2 na tabela · Docling no texto · PyMuPDF de rede de segurança

for pagina in documento:
    regioes = yolo.detectar(pagina)               # DocLayout-YOLO (detecção 9/10)

    if any(r.tipo == "tabela" for r in regioes):
        # Chandra OCR 2 na PÁGINA INTEIRA -> raw com <div data-bbox data-label>
        raw    = chandra.ocr_layout(pagina_hi)    # bbox exata + tipo + <sup>/<sub>
        blocos = parse_data_bbox(raw)             # calibra por (min+max) das coordenadas
    else:
        blocos  = docling.texto_na_ordem(pagina)  # Docling: reading order + fidelidade
        blocos += rede_seguranca_pymupdf(pagina)  # recupera o que o Docling dropou

    for fig in (r for r in regioes if r.tipo == "figura"):
        blocos.append(chandra.descrever(recorte(fig)))   # subtipo: gráfico / foto

    blocos = ordenar_por_coluna(blocos)           # ordem de leitura (2 colunas)
    salvar(blocos)                                 # layout.json + pages/ + document.md
